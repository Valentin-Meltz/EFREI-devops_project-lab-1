# ConfigMap and fake Secret created out of band (never committed); ghcr-pull already exists

$ kubectl create configmap myapp-config --from-literal=APP_ENV=dev -n dev

# 2.1 Apply the base Deployment and Service (kustomization.yaml is not a cluster resource)

$ kubectl apply -f k8s/base/deployment.yaml -f k8s/base/service.yaml -n dev
deployment.apps/myapp created
service/myapp created

$ kubectl rollout status deployment/myapp -n dev --timeout=420s
