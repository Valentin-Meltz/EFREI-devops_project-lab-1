# Ephemeral preview environment: one namespace pr-<number> per Pull Request,
# running the Session 2 application. Each PR has its own state file:
#   terraform init -reconfigure -backend-config="path=state/pr-<number>.tfstate"
# (-reconfigure: switching PRs must never migrate one PR's state into another's).
terraform {
  backend "local" {}

  required_providers {
    kubernetes = {
      source  = "hashicorp/kubernetes"
      version = "~> 2.30"
    }
  }
}

provider "kubernetes" {
  config_path = "~/.kube/config"
  # Pinned, so a preview can never land on whatever context is current.
  config_context = var.kube_context
}

locals {
  namespace = "pr-${var.pr_number}"
  app       = { app = "myapp" }
  # Stamped when the namespace is created (see ignore_changes below).
  expires = formatdate("YYYY-MM-DD", timeadd(plantimestamp(), "${var.ttl_days * 24}h"))
}

resource "kubernetes_namespace" "preview" {
  metadata {
    name = local.namespace
    labels = {
      purpose = "preview"
      pr      = var.pr_number
      # Cost control (4.5): who to ask, and when it may be deleted.
      owner   = var.owner
      expires = local.expires
    }
  }

  lifecycle {
    # The expiry date is set once, at creation; later plans must not move it.
    ignore_changes = [metadata[0].labels["expires"]]
  }
}

# The GHCR package is private, and the kubelet only lets a pod use an image
# pulled with credentials if the pod has credentials too. The token is a
# write-only argument fed by an ephemeral variable: it is sent to Kubernetes
# but never written to the plan or the state.
resource "kubernetes_secret" "ghcr_pull" {
  metadata {
    name      = "ghcr-pull"
    namespace = kubernetes_namespace.preview.metadata[0].name
  }

  type = "kubernetes.io/dockerconfigjson"

  data_wo = {
    ".dockerconfigjson" = jsonencode({
      auths = {
        "ghcr.io" = {
          auth = base64encode("${var.ghcr_user}:${var.ghcr_token}")
        }
      }
    })
  }
  # Write-only values are not compared between runs: increase this number to
  # push a rotated token.
  data_wo_revision = 1
}

resource "kubernetes_deployment" "app" {
  metadata {
    name      = "myapp"
    namespace = kubernetes_namespace.preview.metadata[0].name
    labels    = local.app
  }

  spec {
    replicas = 1

    selector {
      match_labels = local.app
    }

    template {
      metadata {
        labels = local.app
      }

      spec {
        automount_service_account_token = false

        image_pull_secrets {
          name = kubernetes_secret.ghcr_pull.metadata[0].name
        }

        security_context {
          run_as_non_root = true
          run_as_user     = 1000 # "node" in the image; must be numeric for run_as_non_root
          seccomp_profile {
            type = "RuntimeDefault"
          }
        }

        container {
          name              = "myapp"
          image             = var.image
          image_pull_policy = "IfNotPresent"

          port {
            name           = "http"
            container_port = 3000
          }

          env {
            name  = "APP_ENV"
            value = local.namespace
          }

          readiness_probe {
            http_get {
              path = "/health"
              port = "http"
            }
            initial_delay_seconds = 2
            period_seconds        = 5
          }

          liveness_probe {
            http_get {
              path = "/health"
              port = "http"
            }
            initial_delay_seconds = 10
            period_seconds        = 10
          }

          resources {
            requests = {
              cpu    = "50m"
              memory = "64Mi"
            }
            limits = {
              cpu    = "250m"
              memory = "128Mi"
            }
          }

          security_context {
            allow_privilege_escalation = false
            read_only_root_filesystem  = true
            capabilities {
              drop = ["ALL"]
            }
          }
        }
      }
    }
  }

  # A preview that cannot start should fail the pipeline quickly.
  timeouts {
    create = "3m"
    update = "3m"
  }
}

resource "kubernetes_service" "app" {
  metadata {
    name      = "myapp"
    namespace = kubernetes_namespace.preview.metadata[0].name
    labels    = local.app
  }

  spec {
    type     = "ClusterIP"
    selector = local.app

    port {
      name        = "http"
      port        = 80
      target_port = "http"
    }
  }
}

==> infra/preview/variables.tf <==
variable "pr_number" {
  description = "Pull Request number; the namespace is pr-<pr_number>."
  type        = string

  validation {
    condition     = can(regex("^[0-9]+$", var.pr_number))
    error_message = "pr_number must be a Pull Request number (digits only)."
  }
}

variable "image" {
  description = "Image to preview, pinned to a tag or digest (the commit SHA of the PR), never :latest."
  type        = string

  validation {
    condition     = can(regex("[:@][^/]+$", var.image)) && !endswith(var.image, ":latest")
    error_message = "image must carry an explicit tag or digest other than latest."
  }
}

variable "owner" {
  description = "Who is responsible for the preview (for example the PR author); a namespace label."
  type        = string

  validation {
    condition     = can(regex("^[A-Za-z0-9]([-A-Za-z0-9_.]{0,61}[A-Za-z0-9])?$", var.owner))
    error_message = "owner must be a valid Kubernetes label value (63 characters max, letters, digits, '-', '_', '.')."
  }
}

variable "ttl_days" {
  description = "Days until the preview expires; written once to the expires label."
  type        = number
  default     = 3

  validation {
    condition     = var.ttl_days >= 1 && var.ttl_days <= 14
    error_message = "ttl_days must be between 1 and 14."
  }
}

variable "kube_context" {
  description = "kubeconfig context of the cluster that hosts the previews."
  type        = string
  default     = "kind-devops-lab"
}

variable "ghcr_user" {
  description = "GitHub user that owns the GHCR pull token."
  type        = string
  default     = "Valentin-Meltz"
}

variable "ghcr_token" {
  description = "GHCR token with the read:packages scope. Pass it as TF_VAR_ghcr_token; it is never stored."
  type        = string
  sensitive   = true
  ephemeral   = true
}
