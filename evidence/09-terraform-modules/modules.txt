==> infra/modules/webapp/main.tf <==
# A module that uses a provider outside the hashicorp/ namespace must declare
# it, otherwise Terraform looks for hashicorp/docker. The provider itself is
# configured by the calling root module (infra/envs/<env>).
terraform {
  required_providers {
    docker = {
      source  = "kreuzwerker/docker"
      version = "~> 3.0"
    }
  }
}

resource "docker_image" "app" {
  name = var.image
  # Every environment runs on the same Docker host and may use the same image:
  # destroying one environment must not try to delete an image another one
  # still uses.
  keep_locally = true
}

resource "docker_container" "web" {
  name  = "webapp-${var.env_name}"
  image = docker_image.app.image_id

  ports {
    internal = 80
    external = var.external_port
  }
}

==> infra/modules/webapp/variables.tf <==
variable "env_name" {
  description = "Environment name, used in the container name (webapp-<env_name>)."
  type        = string

  validation {
    condition     = can(regex("^[a-z0-9-]+$", var.env_name))
    error_message = "env_name must contain only lowercase letters, digits and dashes."
  }
}

variable "image" {
  description = "Container image to run, with an explicit tag (for example nginx:1.27)."
  type        = string
}

variable "external_port" {
  description = "Host port mapped to port 80 of the container. Must be unique per environment."
  type        = number

  validation {
    condition     = var.external_port >= 1024 && var.external_port <= 65535
    error_message = "external_port must be between 1024 and 65535."
  }
}

==> infra/envs/dev/main.tf <==
terraform {
  required_providers {
    docker = {
      source  = "kreuzwerker/docker"
      version = "~> 3.0"
    }
  }
}

provider "docker" {}

variable "env_name" { type = string }
variable "image" { type = string }
variable "external_port" { type = number }

module "webapp" {
  source        = "../../modules/webapp"
  env_name      = var.env_name
  image         = var.image
  external_port = var.external_port
}

output "app_url" {
  value = module.webapp.app_url
}

output "container_name" {
  value = module.webapp.container_name
}
